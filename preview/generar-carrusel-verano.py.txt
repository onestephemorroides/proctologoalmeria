#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""Genera carrusel 6 slides (1080x1920) para 'Despues del verano'."""

from pathlib import Path
from PIL import Image, ImageDraw, ImageFont

BASE_DIR = Path("/opt/proctologoalmeria")
FOTOS    = BASE_DIR / "Fotos-base"
OUT_DIR  = BASE_DIR / "Videos" / "carrusel-verano"
OUT_DIR.mkdir(parents=True, exist_ok=True)

W, H = 1080, 1920

TEAL = (26, 172, 173)
NAVY = (13, 27, 42)
WHITE = (255, 255, 255)

FONT_DIR = Path("/usr/share/fonts/opentype/montserrat")
FONTS = {
    "bold": FONT_DIR / "Montserrat-Bold.otf",
    "reg":  FONT_DIR / "Montserrat-Medium.otf",
}
_cache = {}


def fnt(size, weight="reg"):
    key = (size, weight)
    if key not in _cache:
        _cache[key] = ImageFont.truetype(str(FONTS[weight]), size)
    return _cache[key]


def text_w(f, text):
    return f.getlength(text)


def wrap_lines(text, f, max_px):
    words, lines, cur = text.split(), [], ""
    for w in words:
        test = (cur + " " + w).strip()
        if text_w(f, test) <= max_px:
            cur = test
        else:
            if cur: lines.append(cur)
            cur = w
    if cur: lines.append(cur)
    return lines


def draw_text(draw, text, y, f, color=WHITE, align="center", max_w=940, x0=80):
    for para in text.split("\n"):
        for line in wrap_lines(para, f, max_w):
            lw = text_w(f, line)
            try:
                bb = f.getbbox(line)
                lh = bb[3] - bb[1]
            except Exception:
                lh = f.size
            x = (W - lw) // 2 if align == "center" else x0
            draw.text((x, y), line, font=f, fill=color)
            y += lh + 10
    return y + 8


def rule(draw, y, w=160, color=TEAL, thickness=4, cx=W // 2):
    draw.rectangle([(cx - w / 2, y), (cx + w / 2, y + thickness)], fill=color)


def logo(canvas, w=220, bottom=H - 80):
    p = FOTOS / "logo-onestep.png"
    if not p.exists():
        return
    logo_img = Image.open(p).convert("RGBA")
    ratio = w / logo_img.width
    logo_img = logo_img.resize((w, int(logo_img.height * ratio)))
    canvas.paste(logo_img, (int((W - w) / 2), int(bottom - logo_img.height)), logo_img)


def fill_canvas(path):
    img = Image.open(path).convert("RGB")
    scale = max(W / img.width, H / img.height)
    img = img.resize((int(img.width * scale), int(img.height * scale)), Image.LANCZOS)
    top = (img.height - H) // 2
    left = (img.width - W) // 2
    img = img.crop((left, top, left + W, top + H))
    return img.convert("RGBA")


def flat_overlay(img, alpha=0.5, color=(0, 0, 0)):
    ov = Image.new("RGBA", img.size, (*color, int(255 * alpha)))
    return Image.alpha_composite(img, ov)


# -- Slides -----------------------------------------------------------------

def slide_01():
    """PORTADA"""
    canvas = fill_canvas(FOTOS / "gancho-A-playa.png")
    canvas = flat_overlay(canvas, 0.45)
    d = ImageDraw.Draw(canvas)
    y = draw_text(d, "ONEstep® · Proctología · Almería", 240, fnt(30, "reg"), (210, 230, 230))
    y += 40
    y = draw_text(d, "Después del verano", y, fnt(76, "bold"), WHITE)
    rule(d, y + 10, w=180)
    y += 44
    draw_text(d, "Molestias \"ahí abajo\"?", y, fnt(46, "reg"), (200, 235, 235))
    logo(canvas, w=220, bottom=H - 90)
    return canvas


def slide_02():
    """HEMORROIDES"""
    canvas = fill_canvas(FOTOS / "frame-verano-hemorroides.png")
    canvas = flat_overlay(canvas, 0.45, color=NAVY)
    d = ImageDraw.Draw(canvas)
    d.rectangle([(0, 0), (W, 200)], fill=(*NAVY, 255))
    draw_text(d, "Hemorroides", 66, fnt(62, "bold"), WHITE)
    d.rectangle([(0, 200), (W, 208)], fill=(*TEAL, 255))
    y = 1550
    draw_text(d, "Ligadura con banda elástica\no esclerosis, sin cirugía",
              y, fnt(44, "reg"), (200, 235, 235), max_w=920)
    return canvas

def slide_06():
    """CIERRE"""
    canvas = fill_canvas(FOTOS / "foto-doctor-2.jpeg")
    d = ImageDraw.Draw(canvas)
    grad = Image.new("L", (1, H), 0)
    start = int(H * 0.36)
    for y in range(H):
        v = 0 if y < start else int(255 * 0.90 * (y - start) / (H - start))
        grad.putpixel((0, y), v)
    grad = grad.resize((W, H))
    ov = Image.new("RGBA", canvas.size, (*NAVY, 0))
    ov.putalpha(grad)
    canvas = Image.alpha_composite(canvas, ov)
    d = ImageDraw.Draw(canvas)
    y = 1010
    y = draw_text(d, "Diagnóstico y tratamiento", y, fnt(58, "bold"), WHITE)
    y = draw_text(d, "en la misma visita", y - 4, fnt(58, "bold"), WHITE)
    rule(d, y + 10, w=180)
    y += 44
    draw_text(d, "Clínica San Pío · Almería · 950 264 245",
              y, fnt(36, "reg"), (195, 235, 235), max_w=960)
    return canvas


SLIDES = [
    (slide_01, "Portada"),
    (slide_02, "Hemorroides"),
    (slide_03, "Fisura anal"),
    (slide_04, "Condilomas"),
    (slide_05, "Pliegues anales"),
    (slide_06, "Cierre ONEstep"),
]


def main():
    for i, (fn, desc) in enumerate(SLIDES, 1):
        canvas = fn().convert("RGB")
        out = OUT_DIR / f"slide-{i:02d}.png"
        canvas.save(out, "PNG")
        print(f"  slide {i} ({desc}) -> {out}")
    print("\nLISTO:", OUT_DIR)


if __name__ == "__main__":
    main()
