#!/usr/bin/env python3
"""Sondeo: cuanta musica hay en archive.org con licencia REALMENTE usable
por una clinica (uso comercial + permite recortar/mezclar)."""
import json, urllib.parse, urllib.request

BASE = "https://archive.org/advancedsearch.php"
GEN = '(subject:"soul" OR subject:"funk" OR subject:"rnb" OR subject:"jazz")'

# Solo estas licencias sirven: permiten uso comercial Y obras derivadas.
LIC = {
    "CC0 / dominio publico": "licenseurl:(*publicdomain*)",
    "CC-BY (atribucion)":    'licenseurl:("http://creativecommons.org/licenses/by/3.0/" OR "http://creativecommons.org/licenses/by/4.0/")',
    "CC-BY-SA":              'licenseurl:("http://creativecommons.org/licenses/by-sa/3.0/" OR "http://creativecommons.org/licenses/by-sa/4.0/")',
}


def q(query, rows=0, fields=("identifier", "title", "licenseurl", "creator")):
    p = {"q": query, "rows": rows, "output": "json"}
    url = BASE + "?" + urllib.parse.urlencode(p) + "".join(
        "&fl[]=" + f for f in fields)
    req = urllib.request.Request(url, headers={"User-Agent": "Mozilla/5.0"})
    return json.load(urllib.request.urlopen(req, timeout=30))["response"]


for nombre, filtro in LIC.items():
    query = f"{GEN} AND mediatype:audio AND {filtro}"
    try:
        r = q(query, rows=3)
        print(f"\n{nombre}: {r['numFound']} pistas")
        for d in r["docs"]:
            print("   -", str(d.get("title", "?"))[:52],
                  "|", str(d.get("creator", "?"))[:22])
    except Exception as e:
        print(f"\n{nombre}: ERROR {e}")
