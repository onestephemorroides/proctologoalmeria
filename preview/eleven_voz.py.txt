#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""Genera voz con ElevenLabs a partir de un voice_id de la libreria compartida.
Uso: eleven_voz.py <nombre> "<texto>" <voice_id> [es-ES añadir a biblioteca primero]
"""
import json, sys, urllib.request
from pathlib import Path

BASE = Path("/opt/proctologoalmeria")
OUT = BASE / "Audio"
OUT.mkdir(parents=True, exist_ok=True)

KEY = None
for l in (BASE / ".env").read_text().splitlines():
    if l.startswith("ELEVENLABS_API_KEY="):
        KEY = l.split("=", 1)[1].strip().strip('"').strip("'")

nombre, texto, voice_id = sys.argv[1], sys.argv[2], sys.argv[3]
H = {"xi-api-key": KEY, "Content-Type": "application/json"}

payload = {
    "text": texto,
    "model_id": "eleven_multilingual_v2",
    "voice_settings": {"stability": 0.5, "similarity_boost": 0.8, "style": 0.3},
}

print(f">>> {nombre} [{voice_id}]", flush=True)
req = urllib.request.Request(
    f"https://api.elevenlabs.io/v1/text-to-speech/{voice_id}",
    data=json.dumps(payload).encode(), headers=H, method="POST")
try:
    data = urllib.request.urlopen(req, timeout=60).read()
except urllib.error.HTTPError as e:
    print("ERROR", e.code, e.read().decode()[:500], flush=True)
    sys.exit(1)

p = OUT / f"{nombre}.mp3"
p.write_bytes(data)
print(f"    OK -> {p} ({len(data)} bytes)", flush=True)
