FONT_DIR = Path("/usr/share/fonts/opentype/montserrat")
FONTS = {
    "bold":  FONT_DIR / "Montserrat-Bold.otf",
    "reg":   FONT_DIR / "Montserrat-Medium.otf",
    "light": FONT_DIR / "Montserrat-Regular.otf" if (FONT_DIR / "Montserrat-Regular.otf").exists()
             else FONT_DIR / "Montserrat-Medium.otf",
}
_cache = {}


def fnt(size, weight="reg"):
    key = (size, weight)
    if key not in _cache:
        _cache[key] = ImageFont.truetype(str(FONTS[weight]), size)
    return _cache[key]


def text_w(f, text):
    return f.getlength(text)


def wrap_lines(text, f, max_px):
    words, lines, cur = text.split(), [], ""
    for w in words:
        test = (cur + " " + w).strip()
        if text_w(f, test) <= max_px:
            cur = test
        else:
            if cur: lines.append(cur)
            cur = w
    if cur: lines.append(cur)
    return lines


def draw_text(draw, text, y, f, color=WHITE, align="center", max_w=940, x0=80, line_h=None):
    line_h = line_h or int(f.size * 128 / 1000)
    for para in text.split("\n"):
        for line in wrap_lines(para, f, max_w):
            lw = text_w(f, line)
            x = (W - lw) / 2 if align == "center" else x0
            draw.text((x, y), line, font=f, fill=color)
            y += line_h
    return y


def rule(draw, y, w=160, color=TEAL, thickness=4, cx=W // 2):
    draw.rectangle([(cx - w / 2, y), (cx + w / 2, y + thickness)], fill=color)


def logo(canvas, w=220, bottom=H - 80):
    p = FOTOS / "logo-onestep.png"
    if not p.exists():
        return
    logo_img = Image.open(p).convert("RGBA")
    ratio = w / logo_img.width
    logo_img = logo_img.resize((w, int(logo_img.height * ratio)))
    canvas.paste(logo_img, (int((W - w) / 2), int(bottom - logo_img.height)), logo_img)


def fill_canvas(path, anchor="center"):
    """Rellena W x H recortando (cover) desde una imagen."""
    img = Image.open(path).convert("RGB")
    scale = max(W / img.width, H / img.height)
    img = img.resize((int(img.width * scale), int(img.height * scale)), Image.LANCZOS)
    if anchor == "top":
        top = 0
    else:
        top = (img.height - H) // 2
    left = (img.width - W) // 2
    img = img.crop((left, top, left + W, top + H))
    return img.convert("RGBA")


def flat_overlay(img, alpha=0.55, color=(0, 0, 0)):
    ov = Image.new("RGBA", img.size, (*color, int(255 * alpha)))
    return Image.alpha_composite(img, ov)


def bottom_gradient(img, opacity=0.88, start_pct=0.38):
    grad = Image.new("L", (1, H), 0)
    start = int(H * start_pct)
    for y in range(H):
        if y < start:
            v = 0
        else:
            v = int(255 * opacity * (y - start) / (H - start))
        grad.putpixel((0, y), v)
    grad = grad.resize((W, H))
    ov = Image.new("RGBA", img.size, (*NAVY, 0))
    ov.putalpha(grad)
    return Image.alpha_composite(img, ov)
