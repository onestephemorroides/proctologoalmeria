#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
imagen_generar.py — genera imágenes vía Replicate (Flux).

Uso:
    imagen_generar.py <nombre> "<prompt>" [modelo]

Modelos disponibles (alias):
    pro    -> black-forest-labs/flux-1.1-pro   (mejor calidad, ~0.04$)
    dev    -> black-forest-labs/flux-dev       (bueno y barato, ~0.03$)
    schnell-> black-forest-labs/flux-schnell   (rápido y muy barato, ~0.003$)

Guarda en Fotos-base/anatomia/<nombre>.png
"""
import json, os, sys, time, urllib.request
from pathlib import Path

BASE = Path("/opt/proctologoalmeria")
OUT = BASE / "Fotos-base" / "anatomia"
OUT.mkdir(parents=True, exist_ok=True)

MODELOS = {
    "pro": "black-forest-labs/flux-1.1-pro",
    "dev": "black-forest-labs/flux-dev",
    "schnell": "black-forest-labs/flux-schnell",
}

TOKEN = None
for l in (BASE / ".env").read_text().splitlines():
    if l.startswith("REPLICATE_API_TOKEN="):
        TOKEN = l.split("=", 1)[1].strip().strip('"').strip("'")
if not TOKEN:
    print("ERROR: falta REPLICATE_API_TOKEN en .env"); sys.exit(1)

H = {"Authorization": f"Bearer {TOKEN}", "Content-Type": "application/json"}


def post(url, payload):
    r = urllib.request.Request(url, data=json.dumps(payload).encode(),
                               headers=H, method="POST")
    return json.load(urllib.request.urlopen(r, timeout=60))


def get(url):
    r = urllib.request.Request(url, headers=H)
    return json.load(urllib.request.urlopen(r, timeout=60))


def generar(nombre, prompt, alias="dev"):
    modelo = MODELOS.get(alias, MODELOS["dev"])
    print(f">>> {nombre}  [{modelo}]", flush=True)

    entrada = {"prompt": prompt, "aspect_ratio": "2:3",
               "output_format": "png", "safety_tolerance": 5}
    if alias == "schnell":
        entrada.pop("safety_tolerance", None)

    try:
        pred = post(f"https://api.replicate.com/v1/models/{modelo}/predictions",
                    {"input": entrada})
    except urllib.error.HTTPError as e:
        cuerpo = e.read().decode()[:400]
        print(f"    ERROR HTTP {e.code}: {cuerpo}", flush=True)
        # safety_tolerance no existe en algunos modelos: reintentar sin él
        if "safety_tolerance" in cuerpo:
            entrada.pop("safety_tolerance", None)
            print("    reintentando sin safety_tolerance...", flush=True)
            pred = post(f"https://api.replicate.com/v1/models/{modelo}/predictions",
                        {"input": entrada})
        else:
            return False

    pid = pred["id"]
    for _ in range(60):
        time.sleep(4)
        d = get(f"https://api.replicate.com/v1/predictions/{pid}")
        est = d["status"]
        if est == "succeeded":
            url = d["output"] if isinstance(d["output"], str) else d["output"][0]
            data = urllib.request.urlopen(url, timeout=120).read()
            p = OUT / f"{nombre}.png"
            p.write_bytes(data)
            print(f"    OK -> {p}  ({len(data)} bytes)", flush=True)
            return True
        if est in ("failed", "canceled"):
            print(f"    {est.upper()}: {str(d.get('error'))[:300]}", flush=True)
            return False
        print(f"    {est}...", flush=True)
    print("    TIMEOUT", flush=True)
    return False


if __name__ == "__main__":
    if len(sys.argv) < 3:
        print(__doc__); sys.exit(0)
    alias = sys.argv[3] if len(sys.argv) > 3 else "dev"
    ok = generar(sys.argv[1], sys.argv[2], alias)
    sys.exit(0 if ok else 1)
